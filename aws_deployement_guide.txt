STEP 3 - Create VPC, 2 public subnets, IGW, route table (no NAT)
~~~bash
set -euo pipefail

export AZ1="$(aws ec2 describe-availability-zones --region "$REGION" --query 'AvailabilityZones[0].ZoneName' --output text)"
export AZ2="$(aws ec2 describe-availability-zones --region "$REGION" --query 'AvailabilityZones[1].ZoneName' --output text)"

export VPC_ID="$(aws ec2 create-vpc --cidr-block "$VPC_CIDR" --query 'Vpc.VpcId' --output text)"
aws ec2 create-tags --resources "$VPC_ID" --tags "Key=Name,Value=${PROJECT_NAME}-vpc"

aws ec2 modify-vpc-attribute --vpc-id "$VPC_ID" --enable-dns-support '{"Value":true}'
aws ec2 modify-vpc-attribute --vpc-id "$VPC_ID" --enable-dns-hostnames '{"Value":true}'

export SUBNET_PUBLIC_A="$(aws ec2 create-subnet --vpc-id "$VPC_ID" --cidr-block "$SUBNET_A_CIDR" --availability-zone "$AZ1" --query 'Subnet.SubnetId' --output text)"
export SUBNET_PUBLIC_B="$(aws ec2 create-subnet --vpc-id "$VPC_ID" --cidr-block "$SUBNET_B_CIDR" --availability-zone "$AZ2" --query 'Subnet.SubnetId' --output text)"

aws ec2 create-tags --resources "$SUBNET_PUBLIC_A" --tags "Key=Name,Value=${PROJECT_NAME}-public-a"
aws ec2 create-tags --resources "$SUBNET_PUBLIC_B" --tags "Key=Name,Value=${PROJECT_NAME}-public-b"

aws ec2 modify-subnet-attribute --subnet-id "$SUBNET_PUBLIC_A" --map-public-ip-on-launch
aws ec2 modify-subnet-attribute --subnet-id "$SUBNET_PUBLIC_B" --map-public-ip-on-launch

export IGW_ID="$(aws ec2 create-internet-gateway --query 'InternetGateway.InternetGatewayId' --output text)"
aws ec2 create-tags --resources "$IGW_ID" --tags "Key=Name,Value=${PROJECT_NAME}-igw"
aws ec2 attach-internet-gateway --internet-gateway-id "$IGW_ID" --vpc-id "$VPC_ID"

export RTB_PUBLIC_ID="$(aws ec2 create-route-table --vpc-id "$VPC_ID" --query 'RouteTable.RouteTableId' --output text)"

aws ec2 create-route --route-table-id "$RTB_PUBLIC_ID" --destination-cidr-block 0.0.0.0/0 --gateway-id "$IGW_ID"

aws ec2 associate-route-table --route-table-id "$RTB_PUBLIC_ID" --subnet-id "$SUBNET_PUBLIC_A"
aws ec2 associate-route-table --route-table-id "$RTB_PUBLIC_ID" --subnet-id "$SUBNET_PUBLIC_B"

aws ecs wait services-stable --cluster "$CLUSTER_NAME" --services "$API_SERVICE_NAME"
~~~
